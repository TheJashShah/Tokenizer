English
-------

The development of artificial intelligence has changed the way people interact with computers. Modern language models can process enormous collections of text, identify statistical patterns, and generate surprisingly coherent responses. However, understanding how these systems represent information internally remains an important research problem.

A tokenizer sits between raw text and a neural network. It converts characters, words, punctuation, and other symbols into a sequence of integer token IDs. Different tokenization strategies can produce dramatically different sequences for the same sentence. Vocabulary size, training data, regular expressions, and the underlying merge rules all influence the final representation.

In 2026, researchers continue to study whether larger language models necessarily develop better representations, or whether some improvements simply emerge from scale. A small experiment with a different corpus can sometimes reveal interesting behavior. For example, multilingual data may cause frequently occurring words in one language to receive very different tokenizations from words in another language.

The quick brown fox jumps over the lazy dog. Machine learning, natural language processing, computer vision, reinforcement learning, and mechanistic interpretability are all active areas of research.

def tokenize_text(text, pattern):
    """Split text into chunks and convert them into UTF-8 bytes."""
    chunks = pattern.findall(text)

    tokens = []
    for chunk in chunks:
        tokens.extend(chunk.encode("utf-8"))

    return tokens


class Tokenizer:
    def __init__(self, vocab_size=50257):
        self.vocab_size = vocab_size
        self.vocab = {}
        self.merges = {}

    def encode(self, text):
        tokens = []
        for byte in text.encode("utf-8"):
            tokens.append(byte)
        return tokens

    def decode(self, tokens):
        data = bytes(tokens)
        return data.decode("utf-8")


def get_stats(ids):
    counts = {}
    for pair in zip(ids, ids[1:]):

public class Tokenizer {
    private final int vocabSize;

    public Tokenizer(int vocabSize) {
        this.vocabSize = vocabSize;
    }

    public int[] encode(String text) {
        byte[] bytes = text.getBytes(java.nio.charset.StandardCharsets.UTF_8);
        int[] tokens = new int[bytes.length];

        for (int i = 0; i < bytes.length; i++) {
            tokens[i] = bytes[i] & 0xFF;
        }

        return tokens;
    }

    public String decode(int[] tokens) {
        byte[] bytes = new byte[tokens.length];

        for (int i = 0; i < tokens.length; i++) {
            bytes[i] = (byte) tokens[i];
        }

        return new String(
            bytes,
            java.nio.charset.StandardCharsets.UTF_8
        );
    }

    public static void main(String[] args) {
        Tokenizer tokenizer = new Tokenizer(50257);
        System.out.println(tokenizer.encode("Hello 🌍"));
    }
}

#include <iostream>
#include <string>
#include <vector>
#include <unordered_map>
#include <utility>

class Tokenizer {
private:
    int vocab_size;

public:
    explicit Tokenizer(int size = 50257)
        : vocab_size(size) {}

    std::vector<unsigned int> encode(const std::string& text) {
        std::vector<unsigned int> tokens;

        for (unsigned char c : text) {
            tokens.push_back(static_cast<unsigned int>(c));
        }

        return tokens;
    }

    std::string decode(
        const std::vector<unsigned int>& tokens
    ) {
        std::string result;

        for (unsigned int token : tokens) {
            result += static_cast<char>(token);
        }

        return result;
    }
};

int main() {
    Tokenizer tokenizer(50257);

    std::string text =
        "Hello, world! Привет! こんにちは!";

    auto tokens = tokenizer.encode(text);

    for (auto token : tokens) {
        std::cout << token << " ";
    }

    std::cout << std::endl;

    return 0;
}

use std::collections::HashMap;

struct Tokenizer {
    vocab_size: usize,
    vocab: HashMap<usize, Vec<u8>>,
}

impl Tokenizer {
    fn new(vocab_size: usize) -> Self {
        Self {
            vocab_size,
            vocab: HashMap::new(),
        }
    }

    fn encode(&self, text: &str) -> Vec<u8> {
        text.as_bytes().to_vec()
    }

    fn decode(&self, tokens: &[u8]) -> String {
        String::from_utf8_lossy(tokens).to_string()
    }
}

fn main() {
    let tokenizer = Tokenizer::new(50257);

    let text = "Hello, world! 🌍";
    let tokens = tokenizer.encode(text);

    println!("{:?}", tokens);

    let decoded = tokenizer.decode(&tokens);
    println!("{}", decoded);
}
